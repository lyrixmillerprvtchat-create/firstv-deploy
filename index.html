-- ====================================================== -- 1st TRUST BANK - PostgreSQL Database Schema -- Transactions Table & Balance Auditing Trigger -- Author: Senior Full-Stack Engineer -- Description: Complete SQL for transaction logging -- and automatic balance change auditing. -- ====================================================== -- ------------------------------------------------------ -- 1. ACCOUNTS TABLE (Reference for foreign keys) -- This is the main accounts table that stores customer -- account information and current balance. -- ------------------------------------------------------ CREATE TABLE IF NOT EXISTS accounts ( id SERIAL PRIMARY KEY, account_number VARCHAR(20) UNIQUE NOT NULL, customer_name VARCHAR(100) NOT NULL, email VARCHAR(100) NOT NULL, phone VARCHAR(20) NOT NULL, address TEXT, country VARCHAR(50), balance DECIMAL(15, 2) DEFAULT 0.00 NOT NULL, is_active BOOLEAN DEFAULT TRUE, created_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() ); -- Index for fast lookups by account number CREATE INDEX IF NOT EXISTS idx_accounts_account_number ON accounts(account_number); CREATE INDEX IF NOT EXISTS idx_accounts_email ON accounts(email); -- ------------------------------------------------------ -- 2. TRANSACTIONS TABLE -- Records every financial transaction (send, receive, -- admin adjustments, opening balance, etc.) -- ------------------------------------------------------ CREATE TABLE IF NOT EXISTS transactions ( id BIGSERIAL PRIMARY KEY, account_number VARCHAR(20) NOT NULL REFERENCES accounts(account_number) ON DELETE CASCADE, transaction_type VARCHAR(30) NOT NULL, -- 'SEND', 'RECEIVE', 'ADMIN_ADJUST', 'OPENING_BALANCE' amount DECIMAL(15, 2) NOT NULL, -- Positive for credits, negative for debits reference VARCHAR(100) NOT NULL, -- Unique transaction reference (e.g., TRF_1234567890) description TEXT, -- Optional description/reason balance_after DECIMAL(15, 2) NOT NULL, -- Account balance immediately after transaction counterparty_account VARCHAR(20), -- For transfers: the other party's account number counterparty_bank VARCHAR(100), -- For transfers: the other party's bank name counterparty_name VARCHAR(100), -- For transfers: receiver/sender name created_at TIMESTAMPTZ DEFAULT NOW(), created_by VARCHAR(50) DEFAULT 'SYSTEM' -- 'SYSTEM', 'ADMIN', or customer identifier ); -- Indexes for fast transaction queries and reporting CREATE INDEX IF NOT EXISTS idx_transactions_account_number ON transactions(account_number); CREATE INDEX IF NOT EXISTS idx_transactions_created_at ON transactions(created_at); CREATE INDEX IF NOT EXISTS idx_transactions_reference ON transactions(reference); CREATE INDEX IF NOT EXISTS idx_transactions_type ON transactions(transaction_type); -- ------------------------------------------------------ -- 3. BALANCE AUDIT LOG TABLE -- Stores historical snapshots of every balance change -- for complete audit trail and regulatory compliance. -- ------------------------------------------------------ CREATE TABLE IF NOT EXISTS balance_audit_log ( id BIGSERIAL PRIMARY KEY, account_number VARCHAR(20) NOT NULL, old_balance DECIMAL(15, 2) NOT NULL, new_balance DECIMAL(15, 2) NOT NULL, amount_change DECIMAL(15, 2) GENERATED ALWAYS AS (new_balance - old_balance) STORED, transaction_reference VARCHAR(100), -- Links to transactions.reference changed_by VARCHAR(50) DEFAULT 'SYSTEM', -- 'SYSTEM', 'ADMIN', or user identifier changed_at TIMESTAMPTZ DEFAULT NOW(), reason TEXT -- Optional reason for adjustment ); -- Index for audit trail queries CREATE INDEX IF NOT EXISTS idx_audit_account_number ON balance_audit_log(account_number); CREATE INDEX IF NOT EXISTS idx_audit_changed_at ON balance_audit_log(changed_at); -- ------------------------------------------------------ -- 4. AUDIT TRIGGER FUNCTION -- Automatically records ANY balance change on the accounts table. -- This trigger fires on UPDATE operations when the balance column changes. -- ------------------------------------------------------ CREATE OR REPLACE FUNCTION audit_balance_change() RETURNS TRIGGER AS $$ BEGIN -- Check if the balance column has actually changed IF (TG_OP = 'UPDATE' AND OLD.balance IS DISTINCT FROM NEW.balance) THEN -- Insert audit record with the before/after balance values INSERT INTO balance_audit_log ( account_number, old_balance, new_balance, changed_by, changed_at, reason ) VALUES ( NEW.account_number, OLD.balance, NEW.balance, COALESCE(current_setting('my.app.current_user', true), 'SYSTEM'), NOW(), 'Automatic trigger: balance updated' ); -- Optional: Raise a notice for debugging (can be commented in production) -- RAISE NOTICE 'Balance audit: Account % changed from % to %', -- NEW.account_number, OLD.balance, NEW.balance; END IF; -- Update the updated_at timestamp on the accounts table NEW.updated_at = NOW(); RETURN NEW; END; $$ LANGUAGE plpgsql; -- ------------------------------------------------------ -- 5. ATTACH TRIGGER TO ACCOUNTS TABLE -- This ensures EVERY balance change is automatically logged -- without requiring application-level code changes. -- ------------------------------------------------------ DROP TRIGGER IF EXISTS trigger_audit_balance ON accounts; CREATE TRIGGER trigger_audit_balance AFTER UPDATE OF balance ON accounts FOR EACH ROW EXECUTE FUNCTION audit_balance_change(); -- ------------------------------------------------------ -- 6. ADDITIONAL HELPER FUNCTIONS & TRIGGERS -- ------------------------------------------------------ -- 6a. Function to automatically create a transaction record -- when a balance change occurs (optional but recommended) CREATE OR REPLACE FUNCTION auto_create_transaction_record() RETURNS TRIGGER AS $$ DECLARE v_reference VARCHAR(100); v_amount_change DECIMAL(15,2); BEGIN -- Only proceed if balance changed IF (TG_OP = 'UPDATE' AND OLD.balance IS DISTINCT FROM NEW.balance) THEN -- Calculate the amount change v_amount_change := NEW.balance - OLD.balance; v_reference := 'AUTO_' || TO_CHAR(NOW(), 'YYYYMMDDHH24MISS') || '_' || LPAD(CAST(NEW.id AS TEXT), 6, '0'); -- Insert into transactions table if the change wasn't already recorded -- (This prevents duplicate transaction entries when both trigger and app insert) IF NOT EXISTS ( SELECT 1 FROM transactions WHERE account_number = NEW.account_number AND reference LIKE 'AUTO_%' AND created_at > NOW() - INTERVAL '1 second' ) THEN INSERT INTO transactions ( account_number, transaction_type, amount, reference, description, balance_after, created_by ) VALUES ( NEW.account_number, CASE WHEN v_amount_change > 0 THEN 'AUTO_CREDIT' WHEN v_amount_change < 0 THEN 'AUTO_DEBIT' ELSE 'AUTO_ADJUST' END, v_amount_change, v_reference, 'Auto-generated from balance change', NEW.balance, 'TRIGGER' ); END IF; END IF; RETURN NEW; END; $$ LANGUAGE plpgsql; -- Optional: Uncomment to attach the auto-transaction trigger -- DROP TRIGGER IF EXISTS trigger_auto_transaction ON accounts; -- CREATE TRIGGER trigger_auto_transaction -- AFTER UPDATE OF balance ON accounts -- FOR EACH ROW -- EXECUTE FUNCTION auto_create_transaction_record(); -- ------------------------------------------------------ -- 7. SAMPLE QUERIES FOR ADMIN & AUDITING -- ------------------------------------------------------ -- View recent balance changes for a specific account -- SELECT * FROM balance_audit_log -- WHERE account_number = '10012345678' -- ORDER BY changed_at DESC LIMIT 20; -- View transaction history for an account -- SELECT * FROM transactions -- WHERE account_number = '10012345678' -- ORDER BY created_at DESC; -- Get total credits and debits per account (daily summary) -- SELECT -- account_number, -- DATE(created_at) as transaction_date, -- SUM(CASE WHEN amount > 0 THEN amount ELSE 0 END) as total_credits, -- SUM(CASE WHEN amount < 0 THEN -amount ELSE 0 END) as total_debits, -- COUNT(*) as transaction_count -- FROM transactions -- GROUP BY account_number, DATE(created_at) -- ORDER BY transaction_date DESC; -- Find accounts with suspicious activity (multiple admin adjustments) -- SELECT account_number, COUNT(*) as adjustment_count -- FROM balance_audit_log -- WHERE changed_by = 'ADMIN' -- AND changed_at > NOW() - INTERVAL '30 days' -- GROUP BY account_number -- HAVING COUNT(*) > 5; -- ------------------------------------------------------ -- 8.
